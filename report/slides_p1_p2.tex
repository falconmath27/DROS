% =============================================================== P2: theory
\section{Optimal stopping}

\begin{frame}{American options are optimal stopping problems}
\begin{itemize}
\item Exercising at date $j$ pays $R_j=e^{-rt_j}(K-S_j)^+$ (discounted to date 0); the price is
  $V_0=\sup_{\tau}\mathbb E^{\mathbb Q}[R_\tau]$ over stopping times $\tau$
  ($\{\tau=j\}\in\mathcal F_j$: decide using the past only).
\item \textbf{Snell envelope:} $U_H=R_H$, \ $U_j=\max\big(R_j,\ \mathbb E[U_{j+1}\mid\mathcal F_j]\big)$;
  smallest supermartingale above $R$; \ $\tau^*=\min\{j:U_j=R_j\}$.
\item Exercise when the payoff beats the \emph{continuation value} $C_j=\mathbb E[U_{j+1}\mid\mathcal F_j]$.
\item \textbf{Baselines} estimate $C_j$ backwards in time: Longstaff--Schwartz (regression),
  Deep Optimal Stopping (one network per date), fitted Q-iteration (bootstrapped targets).
\item Markov prices: $C_j=c_j(S_j)$ needs only today's state. \textbf{With memory (rough
  volatility) it does not.}
\end{itemize}
\end{frame}

\begin{frame}{Free boundary and exact ground truth}
\begin{itemize}
\item $\min\big(-\partial_tV-\mathcal LV,\ V-g\big)=0$: continue where the Black--Scholes /
  Feynman--Kac PDE holds, exercise where $V=g$; smooth pasting $\partial_SV=-1$ at the boundary.
\item Perpetual put: $L^*=\frac{2rK}{2r+\sigma^2}=30$, \ $V(36)=5.79$.
\item Finite $T$: Crank--Nicolson in $\log S$ with a penalty for $V\ge g$.
  \textbf{Tree and PDE agree to $4\times10^{-4}$} on all 20 Longstaff--Schwartz test cases;
  for $S_0=36$, $\sigma=0.2$, $T=1$ the boundary rises from $32.9$ to $K=40$.
\item \textbf{Optional stopping:} for a martingale no rule beats holding,
  so memory can only help when increments are dependent.
\end{itemize}
\end{frame}

% =============================================================== P1: OSPG
\section{OSPG}

\begin{frame}{OSPG: learn the stopping rule directly}
\begin{itemize}
\item Policy outputs $\varphi_j=\mathbb P(\text{stop at }j\mid\text{not yet},\ \mathcal F_j)$
  from the history (GRU) or today only (MLP).
\item Stopping does not change the path, so the stopping-time law is exact:
  \[\psi_j=\varphi_j\prod_{n<j}(1-\varphi_n),\qquad J(\theta)=\mathbb E\Big[\sum_j\psi_jR_j\Big].\]
\item Policy gradient with a constant baseline:
  $\nabla_\theta J=\mathbb E\big[\sum_j(R_j-b)\psi_j\nabla_\theta\log\psi_j\big]$.
\item No rollouts, \textbf{no backward recursion}: no bias carried from date to date.
\item Made to work with an initial ``mostly wait'' bias, a small logit penalty and a separate
  date-0 decision; without them it collapses to ``stop now'' or ``never stop''.
\end{itemize}
\end{frame}
